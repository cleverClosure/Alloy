# GFX-001 arm64ec cross file with the darwin-teb configuration.
# Author: Timur Isaev
#
# Same toolchain as DXMT's in-tree build-arm64ec.txt, plus the two flags that
# retarget the Windows x18=TEB convention for Darwin (result 02 policy):
#   -ffixed-x18        compiler never allocates or reads x18
#   -I darwin-teb-overlay   patched winnt.h shadows the toolchain's; NtCurrentTeb()
#                           compiles to tpidrro_el0 + TSD slot 6 instead of x18
# Kept outside third_party/src/dxmt so the DXMT tree stays zero-change.

strip = 'arm64ec-w64-mingw32-strip'
windres = 'arm64ec-w64-mingw32-windres'

[built-in options]
c_args = ['-ffixed-x18', '-I/Users/cleverclosure/Developer/Alloy/spikes/CPU-001/work/darwin-teb-overlay']
cpp_args = ['-ffixed-x18', '-I/Users/cleverclosure/Developer/Alloy/spikes/CPU-001/work/darwin-teb-overlay']

[properties]

[host_machine]
